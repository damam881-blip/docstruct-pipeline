    def _process_page(
        self,
        doc: fitz.Document,
        input_path: Path,
        page_number: int,
        resolved: ResolvedConfig,
        debug_root: Path,
        output_manager: OutputManager,
    ) -> PageExtractionResult:
        started = time.perf_counter()

        config = resolved.values

        page_debug = (
            debug_root
            / f"page_{page_number:04d}"
        )

        if (
            self.debug
            or config.get(
                "debug",
                {},
            ).get(
                "enabled",
                False,
            )
        ):
            page_debug.mkdir(
                parents=True,
                exist_ok=True,
            )

        analyzer = PDFAnalyzer(config)

        analysis = analyzer.analyze_page(
            doc,
            page_number,
            render_layout=True,
        )

        page = doc[page_number - 1]

        rows: list[ExtractedRow] = []
        issues: list[ValidationIssue] = []
        fallback: list[str] = []
        free_text = ""
        structure_method = "none"
        structure_confidence = 0.0
        structure_candidates: list[dict[str, Any]] = []
        executed_strategy = "unresolved"
        text_detection_result = None
        text_detection_error: str | None = None

        try:
            words = []
            image = None
            preprocessed = None

            # =====================================================
            # OUTPUT / PROCESSING MODE
            # =====================================================

            output_mode = str(
                config.get(
                    "output",
                    {},
                ).get(
                    "mode",
                    "full",
                )
            ).strip().lower()

            crop_only_modes = {
                "cropped_cells_only",
                "cropped_rows_only",
                "cropped_rows_and_cells",
            }

            crop_only = (
                output_mode
                in crop_only_modes
            )

            # =====================================================
            # OCR RECOGNITION INITIALIZATION
            #
            # IMPORTANT:
            # In crop-only modes character recognition is never initialized.
            # A lightweight detection-only CRAFT reader may be initialized
            # later, and only if pure visual geometry cannot find structure.
            # =====================================================

            if crop_only:
                ocr = None

                # Keep the legacy marker for compatibility with existing tests
                # and add the more precise wording for new diagnostics.
                fallback.append("ocr_initialization_skipped")
                fallback.append("ocr_recognition_skipped")

            else:
                ocr = create_ocr_engine(
                    config
                )

            strategy = (
                analysis.recommended_strategy
            )

            digital = DigitalTextExtractor()
            native_pdf_words = []

            # In crop-only mode, digital/hybrid PDFs can still provide native
            # word coordinates from PyMuPDF.  This is geometry metadata, not OCR.
            if crop_only and analysis.page_type in {PageType.DIGITAL, PageType.HYBRID}:
                try:
                    native_pdf_words = digital.extract_words(page)
                except Exception:
                    native_pdf_words = []
                if native_pdf_words:
                    fallback.append("native_digital_word_geometry")

            # =====================================================
            # PAGE ACQUISITION
            # =====================================================

            if (
                strategy
                == Strategy.DIGITAL_TEXT
                and not crop_only
            ):
                # Normal digital PDF extraction.
                words = digital.extract_words(
                    page
                )

                fallback.append(
                    "digital_word_coordinates"
                )
                executed_strategy = "digital_text"

            else:
                # Image-based processing is required for scanned pages and for
                # geometry-only crop modes.
                extractor = ImageExtractor(
                    config
                )

                page_image = (
                    extractor.extract_or_render(
                        doc,
                        page,
                        (
                            page_debug
                            / "01_original.png"
                            if page_debug.exists()
                            else None
                        ),
                    )
                )

                image = page_image.image

                preprocessor = (
                    ImagePreprocessor(
                        config
                    )
                )

                preprocessed = (
                    preprocessor.process(
                        image,
                        (
                            page_debug
                            if page_debug.exists()
                            else None
                        ),
                    )
                )

                fallback.append(
                    f"{page_image.source}_page_image"
                )

                if crop_only:
                    # Recognition stays off.  Pure visual/native geometry is
                    # attempted first; detection-only CRAFT is a later fallback.
                    words = []
                    executed_strategy = "geometry_only"
                    fallback.append("geometry_only_no_page_ocr")

                else:
                    ocr_image = (
                        preprocessed
                        .variants
                        .get(
                            "08_deskewed_gray",
                            preprocessed.selected,
                        )
                    )

                    if ocr is None:
                        raise RuntimeError(
                            "OCR engine unexpectedly "
                            "missing in full extraction mode."
                        )

                    ocr_result = (
                        ocr.recognize(
                            ocr_image,
                            mode=config.get(
                                "ocr",
                                {},
                            ).get(
                                "page_mode",
                                "sparse",
                            ),
                        )
                    )

                    words = (
                        ocr_result.words
                    )
                    executed_strategy = f"{ocr_result.engine}_layout_ocr"

                    fallback.append(
                        f"{ocr_result.engine}_layout_ocr"
                    )

            # =====================================================
            # ENSURE AN IMAGE EXISTS
            # =====================================================

            if image is None:
                scale = (
                    float(
                        config.get(
                            "analysis",
                            {},
                        ).get(
                            "digital_layout_dpi",
                            144,
                        )
                    )
                    / 72.0
                )

                pix = page.get_pixmap(
                    matrix=fitz.Matrix(
                        scale,
                        scale,
                    ),
                    alpha=False,
                )

                image = np.frombuffer(
                    pix.samples,
                    dtype=np.uint8,
                ).reshape(
                    pix.height,
                    pix.width,
                    pix.n,
                )

                image = cv2.cvtColor(
                    image[:, :, :3],
                    cv2.COLOR_RGB2BGR,
                )

                words = self._scale_pdf_words_to_image(
                    words,
                    page,
                    image,
                )

            # Native digital coordinates are scaled only after the crop-only
            # page image exists.  They remain non-OCR geometry evidence.
            if crop_only and native_pdf_words:
                words = self._scale_pdf_words_to_image(
                    native_pdf_words,
                    page,
                    image,
                )
                fallback.append("native_digital_words_scaled_to_image")

            page_bbox = BBox(
                0.0,
                0.0,
                float(image.shape[1]),
                float(image.shape[0]),
            )

            # =====================================================
            # BINARY IMAGE FOR STRUCTURE DETECTION
            # =====================================================

            if preprocessed is not None:
                binary = (
                    preprocessed.selected
                )

            else:
                gray = cv2.cvtColor(
                    image,
                    cv2.COLOR_BGR2GRAY,
                )

                _, binary = cv2.threshold(
                    gray,
                    0,
                    255,
                    cv2.THRESH_BINARY_INV
                    | cv2.THRESH_OTSU,
                )

            # =====================================================
            # ANCHOR SEGMENTATION CONFIG
            #
            # Crop-only:
            #   - force OpenCV projection
            #   - disable cell OCR
            #
            # We use a COPY so normal extraction config
            # is not changed.
            # =====================================================

            anchor_config = config

            if crop_only:
                anchor_config = (
                    copy.deepcopy(config)
                )

                anchor_cfg = (
                    anchor_config
                    .setdefault(
                        "borderless_table",
                        {},
                    )
                    .setdefault(
                        "row_detection",
                        {},
                    )
                    .setdefault(
                        "anchor_segmentation",
                        {},
                    )
                )

                if anchor_cfg.get("enabled", False):
                    anchor_cfg[
                        "candidate_source"
                    ] = "projection"

                anchor_cfg.setdefault(
                    "cell_ocr",
                    {},
                )["enabled"] = False

            anchor_result = (
                AnchorRowSegmenter(
                    anchor_config,
                    ocr,
                ).segment(
                    image,
                    words,
                    input_path.name,
                    page_number,
                    crop_dir=(
                        page_debug
                        / "crops"
                        if page_debug.exists()
                        else None
                    ),
                )
            )

            # =====================================================
            # BUILD ROWS
            # =====================================================

            if crop_only:
                # AnchorRowSegmenter normally creates rows from OCR
                # words. In crop-only there are intentionally no
                # words, so build rows/cells from geometry.
                if anchor_result.anchors_y:
                    rows = (
                        self._rows_from_anchor_geometry(
                            image=image,
                            anchors_y=(
                                anchor_result
                                .anchors_y
                            ),
                            config=config,
                            file_name=(
                                input_path.name
                            ),
                            page_number=(
                                page_number
                            ),
                        )
                    )

                    if rows:
                        structure_method = "opencv_anchor_projection"
                        structure_confidence = anchor_result.confidence
                        fallback.append(
                            "anchor_geometry_"
                            f"{len(anchor_result.anchors_y)}"
                            "_anchors"
                        )

            elif anchor_result.rows:
                rows = (
                    anchor_result.rows
                )

                structure_method = "anchor_row_segmentation"
                structure_confidence = anchor_result.confidence
                fallback.append(
                    "anchor_row_segmentation_"
                    f"{len(anchor_result.anchors_y)}"
                    "_anchors"
                )

            # =====================================================
            # BORDERED STRUCTURE DETECTION
            # =====================================================

            bordered = (
                BorderedTableDetector(
                    config
                ).detect(
                    binary
                )
            )

            if page_debug.exists():
                cv2.imwrite(
                    str(
                        page_debug
                        / "05_horizontal_lines.png"
                    ),
                    bordered.horizontal_mask,
                )

                cv2.imwrite(
                    str(
                        page_debug
                        / "06_vertical_lines.png"
                    ),
                    bordered.vertical_mask,
                )

                region_overlay = (
                    LayoutAnalyzer.draw_regions(
                        image,
                        bordered.regions,
                    )
                )

                cv2.imwrite(
                    str(
                        page_debug
                        / "07_detected_regions.png"
                    ),
                    region_overlay,
                )

            # =====================================================
            # GENERIC STRUCTURE FALLBACKS
            #
            # Evidence is ranked rather than tied to a filename/template:
            #   1) ruled grid geometry
            #   2) native/OCR spatial word geometry
            #   3) visual no-OCR geometry (shaded cells / projection)
            # =====================================================

            structure_image = (
                preprocessed.variants.get("08_deskewed_gray", image)
                if preprocessed is not None
                else image
            )

            visual = VisualStructureDetector(config).detect(
                structure_image,
                binary,
                input_path.name,
                page_number,
            )

            if page_debug.exists():
                for mask_name, mask in visual.debug_masks.items():
                    safe_name = mask_name.replace("/", "_").replace(" ", "_")
                    cv2.imwrite(
                        str(page_debug / f"07a_visual_{safe_name}.png"),
                        mask,
                    )

            if not rows:
                bordered_rows: list[ExtractedRow] = []
                if bordered.regions:
                    bordered_rows = self._rows_from_bordered(
                        words,
                        bordered,
                        input_path.name,
                        page_number,
                        config,
                        structure_image,
                        page_debug,
                        geometry_only=crop_only,
                    )

                borderless_rows: list[ExtractedRow] = []
                if words:
                    borderless = BorderlessTableDetector(config).detect(
                        words,
                        input_path.name,
                        page_number,
                        page_bbox,
                    )
                    borderless_rows = [
                        row
                        for table_rows in borderless.rows_by_table.values()
                        for row in table_rows
                    ]

                visual_rows = visual.rows
                if visual_rows and words and not crop_only:
                    visual_rows = self._assign_words_to_geometry_rows(
                        visual_rows,
                        words,
                    )

                bordered_confidence = max(
                    (region.confidence for region in bordered.regions),
                    default=0.0,
                )
                borderless_confidence = (
                    float(np.mean([row.row_confidence for row in borderless_rows]))
                    if borderless_rows
                    else 0.0
                )

                candidates = [
                    ("bordered_grid", bordered_rows, bordered_confidence),
                    (
                        "native_digital_spatial" if crop_only and words else "borderless_spatial",
                        borderless_rows,
                        borderless_confidence,
                    ),
                    (visual.method, visual_rows, visual.confidence),
                ]
                candidates = [candidate for candidate in candidates if candidate[1]]

                if candidates:
                    scored = []
                    structure_candidates = []
                    for method, candidate_rows, confidence in candidates:
                        metrics = self._geometry_candidate_metrics(
                            candidate_rows,
                            confidence,
                        )
                        score = float(metrics["score"])
                        scored.append(
                            (
                                score,
                                method,
                                candidate_rows,
                                confidence,
                            )
                        )
                        structure_candidates.append(
                            {
                                "method": method,
                                **metrics,
                                "selected": False,
                            }
                        )

                    score, structure_method, rows, structure_confidence = max(
                        scored, key=lambda item: item[0]
                    )
                    for candidate_info in structure_candidates:
                        candidate_info["selected"] = (
                            candidate_info["method"] == structure_method
                            and abs(float(candidate_info["score"]) - score) < 1e-9
                        )
                    fallback.append(f"structure_{structure_method}_score={score:.2f}")

            # =====================================================
            # DETECTION-ONLY TEXT-BOX FALLBACK (CROP-ONLY)
            #
            # This is not character recognition.  EasyOCR/CRAFT returns only
            # bounding boxes that say "text exists here".  The spatial boxes
            # are clustered into rows/columns/cells and the cell values remain
            # blank.  It is invoked lazily only when visual/native geometry has
            # not already produced a usable structure.
            # =====================================================

            text_detection_cfg = (
                config.get("visual_structure", {})
                .get("text_detection_fallback", {})
            )
            text_detection_enabled = (
                crop_only and bool(text_detection_cfg.get("enabled", True))
            )
            if not rows and text_detection_enabled:
                try:
                    detector = create_text_detector(config)
                    text_detection_result = detector.detect(structure_image)
                    detected_boxes = text_detection_result.boxes
                    minimum_boxes = max(1, int(text_detection_cfg.get("minimum_boxes", 4)))
                    fallback.append(
                        f"text_detection_only_{text_detection_result.engine}_"
                        f"{text_detection_result.device}_{len(detected_boxes)}_boxes"
                    )
                    log(
                        self.logger,
                        "text_detection_fallback",
                        stage="structure_detection",
                        file_name=input_path.name,
                        page_number=page_number,
                        engine=text_detection_result.engine,
                        device=text_detection_result.device,
                        recognizer_loaded=False,
                        boxes_detected=len(detected_boxes),
                    )

                    if page_debug.exists() and detected_boxes:
                        self._save_ocr_overlay(
                            structure_image,
                            detected_boxes,
                            page_debug / "07c_text_detection_boxes.png",
                        )

                    if len(detected_boxes) >= minimum_boxes:
                        detected_structure = BorderlessTableDetector(config).detect(
                            detected_boxes,
                            input_path.name,
                            page_number,
                            page_bbox,
                        )
                        detected_rows = [
                            row
                            for table_rows in detected_structure.rows_by_table.values()
                            for row in table_rows
                        ]
                        if detected_rows:
                            detection_confidence = max(
                                (region.confidence for region in detected_structure.regions),
                                default=float(text_detection_cfg.get("structural_confidence", 0.60)),
                            )
                            rows = self._prepare_detection_only_rows(
                                detected_rows,
                                detection_confidence,
                            )
                            structure_method = "easyocr_detection_spatial"
                            structure_confidence = detection_confidence
                            executed_strategy = "geometry_plus_text_detection"
                            fallback.append("structure_recovered_by_text_detection")
                except Exception as exc:
                    text_detection_error = f"{type(exc).__name__}: {exc}"
                    fallback.append("text_detection_fallback_failed")
                    issues.append(
                        ValidationIssue(
                            code="TEXT_DETECTION_FALLBACK_FAILED",
                            message=(
                                "Detection-only text-box fallback could not run: "
                                f"{text_detection_error}"
                            ),
                            severity="warning",
                            page_number=page_number,
                        )
                    )

            # Save a visual-region overlay independently from the classic ruled
            # detector so a developer can see why geometry-only cropping worked.
            if page_debug.exists() and visual.regions:
                visual_overlay = LayoutAnalyzer.draw_regions(
                    structure_image,
                    visual.regions,
                )
                cv2.imwrite(
                    str(page_debug / "07b_visual_detected_regions.png"),
                    visual_overlay,
                )

            # =====================================================
            # FREE TEXT FALLBACK
            # =====================================================

            if not rows:
                if crop_only:
                    # Do not create an artificial OCR row.  A geometry-only run
                    # that found no croppable structure is completed, but it is
                    # explicitly reported as a warning rather than a silent success.
                    fallback.append(
                        "geometry_only_"
                        "no_rows_detected"
                    )
                    issues.append(
                        ValidationIssue(
                            code="NO_STRUCTURE_DETECTED",
                            message=(
                                "Crop-only processing completed but no row/cell "
                                "structure could be inferred from page geometry."
                            ),
                            severity="warning",
                            page_number=page_number,
                        )
                    )

                else:
                    free_text = " ".join(
                        w.text
                        for w in words
                    )

                    rows = (
                        self._free_text_row(
                            words,
                            input_path.name,
                            page_number,
                        )
                    )

                    fallback.append(
                        "full_page_text_review"
                    )

            # =====================================================
            # NORMALIZATION / VALIDATION
            # =====================================================

            if crop_only:
                # No semantic OCR validation runs here, but structural warnings
                # (for example NO_STRUCTURE_DETECTED) must be preserved.
                pass

            else:
                reconstructor = (
                    DataReconstructor(
                        config
                    )
                )

                rows = (
                    reconstructor
                    .normalize_rows(
                        rows
                    )
                )

                validator = (
                    Validator(
                        config
                    )
                )

                issues = (
                    validator.validate(
                        rows
                    )
                )

            # =====================================================
            # SAVE CROPS
            # =====================================================

            crop_image = structure_image

            output_manager.save_page_crops(
                crop_image,
                rows,
                page_number,
            )

            processing_seconds = (
                time.perf_counter()
                - started
            )

            # =====================================================
            # PAGE COMPLETION STATUS
            # =====================================================

            issue_severities = {str(issue.severity).lower() for issue in issues}
            if "error" in issue_severities:
                page_status = "completed_with_validation_errors"
            elif issues:
                page_status = "completed_with_warning"
            else:
                page_status = "completed"

            # =====================================================
            # DEBUG OUTPUTS
            # =====================================================

            if page_debug.exists():
                self._save_structure_overlays(
                    structure_image,
                    rows,
                    issues,
                    page_debug,
                )

                # No meaningless OCR overlay in crop-only.
                if not crop_only:
                    self._save_ocr_overlay(
                        structure_image,
                        words,
                        page_debug
                        / "11_ocr_overlay.png",
                    )

                page_report = {
                    "analysis": (
                        dataclass_to_dict(
                            analysis
                        )
                    ),
                    "page_status": page_status,
                    "processing_mode": (
                        "geometry_plus_text_detection"
                        if crop_only and bool(
                            config.get("visual_structure", {})
                            .get("text_detection_fallback", {})
                            .get("enabled", True)
                        )
                        else ("geometry_only" if crop_only else "full_extraction")
                    ),
                    "ocr_enabled": (not crop_only),
                    "ocr_recognition_enabled": (not crop_only),
                    "text_detection_fallback": {
                        "enabled": crop_only and bool(
                            config.get("visual_structure", {})
                            .get("text_detection_fallback", {})
                            .get("enabled", True)
                        ),
                        "used": text_detection_result is not None,
                        "engine": (
                            text_detection_result.engine
                            if text_detection_result is not None
                            else None
                        ),
                        "device": (
                            text_detection_result.device
                            if text_detection_result is not None
                            else None
                        ),
                        "boxes_detected": (
                            len(text_detection_result.boxes)
                            if text_detection_result is not None
                            else 0
                        ),
                        "recognizer_loaded": False,
                        "error": text_detection_error,
                    },
                    "output_mode": (
                        output_mode
                    ),
                    "recommended_strategy": analysis.recommended_strategy.value,
                    "executed_strategy": executed_strategy,
                    "structure_method": structure_method,
                    "structure_confidence": structure_confidence,
                    "structure_detected": bool(rows),
                    "structure_candidates": structure_candidates,
                    "visual_structure": {
                        "method": visual.method,
                        "confidence": visual.confidence,
                        "measurements": visual.measurements,
                    },
                    "anchor_segmentation": {
                        "reason": (
                            anchor_result.reason
                        ),
                        "anchor_count": (
                            len(
                                anchor_result
                                .anchors_y
                            )
                        ),
                        "anchor_y": (
                            anchor_result
                            .anchors_y
                        ),
                        "roi_bbox": (
                            anchor_result
                            .roi_bbox
                            .to_list()
                        ),
                        "confidence": (
                            anchor_result
                            .confidence
                        ),
                    },
                    "rows": [
                        dataclass_to_dict(r)
                        for r in rows
                    ],
                    "issues": [
                        dataclass_to_dict(i)
                        for i in issues
                    ],
                    "fallback_chain": (
                        fallback
                    ),
                    "processing_seconds": (
                        processing_seconds
                    ),
                    "configuration_source": {
                        "base_yaml": (
                            resolved.config_path
                        ),
                        "run_config_json": (
                            resolved.run_config_path
                        ),
                        "page_number": (
                            resolved.page_number
                        ),
                        "page_overrides": (
                            resolved.page_overrides
                        ),
                        "cli_overrides": (
                            resolved.cli_overrides
                        ),
                    },
                    "resolved_config": (
                        resolved.values
                    ),
                }

                (
                    page_debug
                    / "page_report.json"
                ).write_text(
                    json.dumps(
                        page_report,
                        ensure_ascii=False,
                        indent=2,
                        default=str,
                    ),
                    encoding="utf-8",
                )

            result = PageExtractionResult(
                analysis,
                rows,
                free_text,
                issues,
                (
                    str(page_debug)
                    if page_debug.exists()
                    else None
                ),
                processing_seconds,
                fallback,
                True,
                None,
            )

            log(
                self.logger,
                "page_processed",
                stage="extract",
                file_name=(
                    input_path.name
                ),
                page_number=(
                    page_number
                ),
                page_type=(
                    analysis.page_type.value
                ),
                layout_type=(
                    analysis.layout_type.value
                ),
                strategy=(
                    analysis
                    .recommended_strategy
                    .value
                ),
                recommended_strategy=analysis.recommended_strategy.value,
                executed_strategy=executed_strategy,
                processing_mode=(
                    "geometry_plus_text_detection"
                    if crop_only and bool(
                        config.get("visual_structure", {})
                        .get("text_detection_fallback", {})
                        .get("enabled", True)
                    )
                    else ("geometry_only" if crop_only else "full_extraction")
                ),
                output_mode=output_mode,
                ocr_enabled=not crop_only,
                ocr_recognition_enabled=not crop_only,
                text_detection_used=text_detection_result is not None,
                text_detection_boxes=(
                    len(text_detection_result.boxes)
                    if text_detection_result is not None
                    else 0
                ),
                text_detection_device=(
                    text_detection_result.device
                    if text_detection_result is not None
                    else None
                ),
                structure_method=structure_method,
                structure_confidence=structure_confidence,
                structure_detected=bool(rows),
                structure_candidates=[
                    {
                        "method": candidate.get("method"),
                        "score": candidate.get("score"),
                        "selected": candidate.get("selected", False),
                    }
                    for candidate in structure_candidates
                ],
                page_status=page_status,
                rows_detected=(
                    len(rows)
                ),
                columns_detected=max(
                    (
                        len(r.cells)
                        for r in rows
                    ),
                    default=0,
                ),
                ocr_confidence=(
                    0.0
                    if crop_only
                    else (
                        (
                            sum(r.row_confidence for r in rows) / len(rows)
                        )
                        if rows
                        else 0.0
                    )
                ),
                duration_seconds=(

    def _rows_from_anchor_geometry(
        self,
        image: np.ndarray,
        anchors_y: list[float],
        config: dict[str, Any],
        file_name: str,
        page_number: int,
        table_id: str = "table_01",
    ) -> list[ExtractedRow]:
        """
        Build rows and cells entirely from geometry.

        This method is used by crop-only modes and therefore
        does not require OCR words or an OCR engine.
        """

        if not anchors_y:
            return []

        h, w = image.shape[:2]

        anchor_cfg = (
            config.get(
                "borderless_table",
                {},
            )
            .get(
                "row_detection",
                {},
            )
            .get(
                "anchor_segmentation",
                {},
            )
        )

        roi_ratio = anchor_cfg.get(
            "roi_ratio",
            [
                0.0,
                0.25,
                0.20,
                0.90,
            ],
        )

        ry0 = int(
            float(roi_ratio[2])
            * h
        )

        ry1 = int(
            float(roi_ratio[3])
            * h
        )

        anchor_y = sorted(
            float(v)
            for v in anchors_y
        )

        gaps = [
            b - a
            for a, b
            in zip(
                anchor_y,
                anchor_y[1:],
            )
        ]

        typical_gap = (
            float(
                np.median(gaps)
            )
            if gaps
            else float(
                anchor_cfg.get(
                    "default_row_height_px",
                    80,
                )
            )
        )

        boundary_mode = str(
            anchor_cfg.get(
                "boundary_mode",
                "midpoint",
            )
        ).strip().lower()

        if boundary_mode == "midpoint":
            top = max(
                float(ry0),
                anchor_y[0]
                - typical_gap / 2.0,
            )

            bottom = min(
                float(ry1),
                anchor_y[-1]
                + typical_gap / 2.0,
            )

            boundaries = (
                [top]
                + [
                    (
                        current_anchor
                        + next_anchor
                    )
                    / 2.0
                    for (
                        current_anchor,
                        next_anchor,
                    )
                    in zip(
                        anchor_y,
                        anchor_y[1:],
                    )
                ]
                + [bottom]
            )

        elif boundary_mode == "anchor_start":
            top = float(ry0)
            bottom = float(ry1)

            boundary_offset_px = float(
                anchor_cfg.get(
                    "boundary_offset_px",
                    0.0,
                )
            )

            internal_boundaries = [
                min(
                    max(
                        float(anchor)
                        + boundary_offset_px,
                        top,
                    ),
                    bottom,
                )
                for anchor
                in anchor_y[1:]
            ]

            boundaries = (
                [top]
                + internal_boundaries
                + [bottom]
            )

        else:
            raise ValueError(
                "Unsupported anchor "
                "boundary_mode: "
                f"{boundary_mode!r}. "
                "Allowed values are "
                "'midpoint' and "
                "'anchor_start'."
            )

        column_ratios = (
            anchor_cfg.get(
                "column_boundaries_ratio"
            )
        )

        if not column_ratios:
            expected = max(
                1,
                int(
                    anchor_cfg.get(
                        "expected_columns",
                        1,
                    )
                ),
            )

            table_x0 = float(
                anchor_cfg.get(
                    "table_x0_ratio",
                    0.0,
                )
            )

            table_x1 = float(
                anchor_cfg.get(
                    "table_x1_ratio",
                    1.0,
                )
            )

            column_ratios = [
                table_x0
                + (
                    table_x1
                    - table_x0
                )
                * i
                / expected
                for i
                in range(
                    expected + 1
                )
            ]

        column_boundaries = [
            min(
                float(w),
                max(
                    0.0,
                    float(ratio)
                    * w,
                ),
            )
            for ratio
            in column_ratios
        ]

        column_boundaries = (
            sorted(
                set(
                    column_boundaries
                )
            )
        )

        if len(
            column_boundaries
        ) < 2:
            return []

        rows: list[
            ExtractedRow
        ] = []

        for (
            row_index,
            (
                band_top,
                band_bottom,
            ),
        ) in enumerate(
            zip(
                boundaries,
                boundaries[1:],
            )
        ):
            if (
                band_bottom
                <= band_top
            ):
                continue

            cells: list[
                ExtractedCell
            ] = []

            for (
                col_index,
                (
                    col_left,
                    col_right,
                ),
            ) in enumerate(
                zip(
                    column_boundaries,
                    column_boundaries[1:],
                )
            ):
                if (
                    col_right
                    <= col_left
                ):
                    continue

                cell_bbox = BBox(
                    col_left,
                    band_top,
                    col_right,
                    band_bottom,
                ).clamped(
                    w,
                    h,
                )

                cells.append(
                    ExtractedCell(
                        file_name=(
                            file_name
                        ),
                        page_number=(
                            page_number
                        ),
                        table_id=(
                            table_id
                        ),
                        row_index=(
                            row_index
                        ),
                        column_index=(
                            col_index
                        ),
                        bbox=(
                            cell_bbox
                        ),
                        crop_path=None,
                        engine="none",
                        confidence=0.0,
                        raw_value="",
                        normalized_value="",
                    )
                )

            if not cells:
                continue

            row_bbox = BBox(
                column_boundaries[0],
                band_top,
                column_boundaries[-1],
                band_bottom,
            ).clamped(
                w,
                h,
            )

            rows.append(
                ExtractedRow(
                    file_name=(
                        file_name
                    ),
                    page_number=(
                        page_number
                    ),
                    table_id=(
                        table_id
                    ),
                    row_index=(
                        row_index
                    ),
                    cells=cells,
                    bbox=row_bbox,
                    row_confidence=0.0,
                )
            )

        return rows

    def segment(
        self,
        image: np.ndarray,
        full_page_words: list[WordBox],
        file_name: str,
        page_number: int,
        table_id: str = "table_01",
        crop_dir: Path | None = None,
    ) -> AnchorSegmentationResult:
        if not self.config.get("enabled", False):
            return AnchorSegmentationResult([], [], BBox(0, 0, 0, 0), 0.0, "disabled")
        h, w = image.shape[:2]
        roi_ratio = self.config.get("roi_ratio", [0.0, 0.25, 0.20, 0.90])
        x0, x1 = int(float(roi_ratio[0]) * w), int(float(roi_ratio[1]) * w)
        y0, y1 = int(float(roi_ratio[2]) * h), int(float(roi_ratio[3]) * h)
        roi_bbox = BBox(x0, y0, x1, y1).clamped(w, h)
        rx0, ry0, rx1, ry1 = roi_bbox.to_int()
        roi = image[ry0:ry1, rx0:rx1]
        if roi.size == 0:
            return AnchorSegmentationResult([], [], roi_bbox, 0.0, "empty_roi")

        mode = self.config.get("ocr_mode", "sparse")
        anchor_type = self.config.get("anchor_type", "amount")
        candidate_source = self.config.get("candidate_source", "roi_ocr")
        projection_anchor_y: list[float] | None = None
        if candidate_source == "projection":
            projection_anchor_y = self._projection_anchors(roi, ry0)
            candidate_words = []
            result_confidence = float(self.config.get("projection_confidence", 0.70))
            result_engine = "opencv_projection"
        elif candidate_source == "full_page_words":
            candidate_words = [
                word for word in full_page_words
                if roi_bbox.x0 <= word.bbox.center[0] <= roi_bbox.x1
                and roi_bbox.y0 <= word.bbox.center[1] <= roi_bbox.y1
            ]
            result_confidence = float(np.mean([word.confidence for word in candidate_words])) if candidate_words else 0.0
            result_engine = candidate_words[0].source if candidate_words else "none"
        else:
            result = self.ocr_engine.recognize(roi, mode=mode, numeric=anchor_type == "amount")
            candidate_words = [
                WordBox(
                    word.text,
                    BBox(word.bbox.x0 + rx0, word.bbox.y0 + ry0, word.bbox.x1 + rx0, word.bbox.y1 + ry0),
                    word.confidence,
                    word.source,
                    word.block_no,
                    word.line_no,
                    word.word_no,
                    word.language,
                )
                for word in result.words
            ]
            result_confidence = result.confidence
            result_engine = result.engine

        candidates: list[tuple[float, float, str]] = []
        minimum_confidence = float(self.config.get("minimum_confidence", 0.15))
        minimum_digits = int(self.config.get("minimum_digits", 1))
        for word in candidate_words:
            text = word.text.strip()
            if word.confidence < minimum_confidence:
                continue
            normalized = DataReconstructor.normalize_digits(text)
            digit_count = sum(ch.isdigit() for ch in normalized)
            valid = False
            if anchor_type == "amount":
                valid = DataReconstructor.parse_amount(text) is not None and digit_count >= minimum_digits
            elif anchor_type == "date":
                valid = digit_count >= max(4, minimum_digits) and any(sep in normalized for sep in ("/", "-"))
            elif anchor_type == "digit_line":
                # Useful when OCR preserves the numeric line position but confuses
                # Arabic-Indic digits with Latin letters or drops date separators.
                valid = digit_count >= minimum_digits
            if valid:
                candidates.append((word.bbox.center[1], word.confidence, text))

        minimum_gap = float(self.config.get("minimum_anchor_gap_px", max(12, h * 0.008)))
        if projection_anchor_y is not None:
            anchor_y = projection_anchor_y
        else:
            anchors: list[list[tuple[float, float, str]]] = []
            for candidate in sorted(candidates):
                if not anchors or candidate[0] - np.mean([c[0] for c in anchors[-1]]) > minimum_gap:
                    anchors.append([candidate])
                else:
                    anchors[-1].append(candidate)
            anchor_y = [float(np.mean([c[0] for c in group])) for group in anchors]
        min_anchors = int(self.config.get("minimum_anchors", 3))
        if len(anchor_y) < min_anchors:
            return AnchorSegmentationResult([], anchor_y, roi_bbox, result_confidence, f"only_{len(anchor_y)}_anchors")

        gaps = [
            next_anchor - current_anchor
            for current_anchor, next_anchor in zip(
                anchor_y,
                anchor_y[1:],
            )
        ]
        typical_gap = (
            float(median(gaps))
            if gaps
            else float(self.config.get("default_row_height_px", 80))
        )

        # Controls how detected anchors are converted into row boundaries.
        boundary_mode = str(
            self.config.get("boundary_mode", "midpoint")
        ).strip().lower()

        if boundary_mode == "midpoint":
            # Use this mode when each anchor is approximately in the middle
            # of its logical row.
            top = max(float(ry0), anchor_y[0] - typical_gap / 2.0)
            bottom = min(float(ry1), anchor_y[-1] + typical_gap / 2.0)
            boundaries = [top] + [
                (current_anchor + next_anchor) / 2.0
                for current_anchor, next_anchor in zip(anchor_y, anchor_y[1:])
            ] + [bottom]
        elif boundary_mode == "anchor_start":
            # Use this mode when every anchor represents the beginning
            # of a new logical row.
            top = float(ry0)
            bottom = float(ry1)

            # Optional adjustment between the detected anchor center
            # and the actual beginning of the logical row.
            boundary_offset_px = float(self.config.get("boundary_offset_px", 0.0))

            internal_boundaries = [
                min(max(float(anchor) + boundary_offset_px, top), bottom)
                for anchor in anchor_y[1:]
            ]
            boundaries = [top] + internal_boundaries + [bottom]
        else:
            raise ValueError(
                "Unsupported anchor boundary_mode: "
                f"{boundary_mode!r}. "
                "Allowed values are 'midpoint' and 'anchor_start'."
            )

        column_ratios = self.config.get("column_boundaries_ratio")
        if not column_ratios:
            expected = int(self.config.get("expected_columns", 1))
            table_x0 = float(self.config.get("table_x0_ratio", 0.0))
            table_x1 = float(self.config.get("table_x1_ratio", 1.0))
            column_ratios = [table_x0 + (table_x1 - table_x0) * i / expected for i in range(expected + 1)]
        column_boundaries = [min(float(w), max(0.0, float(r) * w)) for r in column_ratios]
        column_boundaries = sorted(set(column_boundaries))

        rows: list[ExtractedRow] = []
        for row_index, (band_top, band_bottom) in enumerate(zip(boundaries, boundaries[1:])):
            row_words = [
                word for word in full_page_words
                if band_top <= word.bbox.center[1] < band_bottom
                and float(self.config.get("table_x0_ratio", 0.0)) * w <= word.bbox.center[0] <= float(self.config.get("table_x1_ratio", 1.0)) * w
            ]
            cells: list[ExtractedCell] = []
            for col_index, (col_left, col_right) in enumerate(zip(column_boundaries, column_boundaries[1:])):
                cell_words = [word for word in row_words if col_left <= word.bbox.center[0] < col_right]
                cell_words = self._sort_words(cell_words)
                text = " ".join(word.text for word in cell_words).strip()
                # A cell crop must represent the configured logical column, not
                # only the tight union of OCR words. This keeps empty space,
                # punctuation, and weakly recognized digits inside the crop.
                bbox = BBox(col_left, band_top, col_right, band_bottom).clamped(w, h)
                confidence = float(np.mean([word.confidence for word in cell_words])) if cell_words else 0.0
                engine = cell_words[0].source if cell_words else result_engine
                crop_path: str | None = None
                refined = self._refine_cell_with_crop_ocr(
                    image=image,
                    col_index=col_index,
                    col_left=col_left,
                    col_right=col_right,
                    band_top=band_top,
                    band_bottom=band_bottom,
                    existing_text=text,
                    existing_confidence=confidence,
                    row_index=row_index,
                    page_number=page_number,
                    table_id=table_id,
                    crop_dir=crop_dir,
                )
                if refined is not None:
                    text, confidence, engine, crop_path = refined
                    bbox = BBox(col_left, band_top, col_right, band_bottom).clamped(w, h)
                cells.append(
                    ExtractedCell(
                        file_name=file_name,
                        page_number=page_number,
                        table_id=table_id,
                        row_index=row_index,
                        column_index=col_index,
                        bbox=bbox,
                        crop_path=crop_path,
                        engine=engine,
                        confidence=confidence,
                        raw_value=text,
                        normalized_value=text,
                    )
                )
            if any(cell.raw_value for cell in cells):
                row_bbox = BBox(
                    column_boundaries[0], band_top, column_boundaries[-1], band_bottom
                ).clamped(w, h)
                nonempty_conf = [cell.confidence for cell in cells if cell.raw_value]
                rows.append(
                    ExtractedRow(
                        file_name=file_name,
                        page_number=page_number,
                        table_id=table_id,
                        row_index=row_index,
                        cells=cells,
                        bbox=row_bbox,
                        row_confidence=float(np.mean(nonempty_conf)) if nonempty_conf else 0.0,
                    )
                )
        return AnchorSegmentationResult(rows, anchor_y, roi_bbox, result_confidence, "ok")

    def _projection_anchors(self, roi: np.ndarray, y_offset: int) -> list[float]:
        """Detect one anchor per text line from a narrow date/amount column.

        Long vertical rules are removed before computing the horizontal ink
        projection. Nearby bands are merged so a two-line date/time cell still
        represents one transaction row.
        """
        gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY) if roi.ndim == 3 else roi.copy()
        _, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY_INV | cv2.THRESH_OTSU)
        vertical_length = max(20, int(binary.shape[0] * float(self.config.get("projection_vertical_kernel_ratio", 0.05))))
        vertical = cv2.morphologyEx(binary, cv2.MORPH_OPEN, cv2.getStructuringElement(cv2.MORPH_RECT, (1, vertical_length)))
        clean = cv2.subtract(binary, vertical)
        close_width = max(1, int(self.config.get("projection_close_width_px", 5)))
        close_height = max(1, int(self.config.get("projection_close_height_px", 2)))
        clean = cv2.morphologyEx(clean, cv2.MORPH_CLOSE, cv2.getStructuringElement(cv2.MORPH_RECT, (close_width, close_height)))
        projection = np.count_nonzero(clean, axis=1)
        threshold_ratio = float(self.config.get("projection_ink_threshold_ratio", 0.02))
        threshold = max(int(self.config.get("projection_min_ink_px", 2)), int(clean.shape[1] * threshold_ratio))
        active = projection > threshold
        minimum_band_height = int(self.config.get("projection_min_band_height_px", 3))
        bands: list[tuple[int, int]] = []
        start: int | None = None
        for index, is_active in enumerate(active):
            if is_active and start is None:
                start = index
            elif not is_active and start is not None:
                if index - start >= minimum_band_height:
                    bands.append((start, index - 1))
                start = None
        if start is not None and len(active) - start >= minimum_band_height:
            bands.append((start, len(active) - 1))

        merge_gap = int(self.config.get("projection_merge_gap_px", self.config.get("minimum_anchor_gap_px", 40)))
        merged: list[tuple[int, int]] = []
        for band in bands:
            if merged and band[0] - merged[-1][1] <= merge_gap:
                merged[-1] = (merged[-1][0], band[1])
            else:
                merged.append(band)
        return [float(y_offset + (start + end) / 2.0) for start, end in merged]
